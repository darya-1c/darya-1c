Загрузка сотрудников в 1С

📌 Что делает эта обработка
Загружает сотрудников в справочник «Сотрудники», изменяет склад у документа, осуществляет поиск, помечает на удаление или безвозвратно удаляет выбранный документ или сотрудника .

⚙️ Как работает
1. Пользователь открывает обработку, нажимает на кнопку «Добавить» и вводит ФИО, возраст и пол сотрудника
2. При нажатии на кнопку «Загрузить» обработка читает данные, проверяет на наличие такого сотрудника в справочнике «Сотрудники»
3. Если есть - выдает соответствующее сообщение, если нет — создает нового сотрудника
4. Также пользователь может может выбрать сотрудника и при нажатии кнопки «Пометить сотрудника на удаление» - пометить на удаление, при нажатии кнопки «Удалить сотрудника» - удалить сотрудника.
5. При переходе на вкладку «Документы» есть возможность изменить склад документа, найти документ, пометить на удаление или удалить безвозвратно

📸 Код модуля формы

//код для кнопки загрузить
Процедура ЗагрузитьНаСервере()
	Для каждого СтрокаТЧ Из Объект.Сотрудники Цикл
		ПроверкаСотрудника = Справочники.Сотрудники.НайтиПоНаименованию(СтрокаТЧ.ФИО);
		Если ПроверкаСотрудника = Справочники.Сотрудники.ПустаяСсылка() Тогда
			НовыйЭлементСправочника = Справочники.Сотрудники.СоздатьЭлемент();
			НовыйЭлементСправочника.Наименование = СтрокаТЧ.ФИО;	
			НовыйЭлементСправочника.Возраст = СтрокаТЧ.Возраст;
			НовыйЭлементСправочника.Пол = СтрокаТЧ.Пол;  
			НовыйЭлементСправочника.Записать();  
		Иначе 
			Сообщение = Новый СообщениеПользователю;
			Сообщение.Текст = "Сотрудник " + СтрокаТЧ.ФИО + " уже существует";
			Сообщение.Сообщить();
		КонецЕсли;
	КонецЦикла;	
КонецПроцедуры

&НаКлиенте
Процедура Загрузить(Команда)
	ЗагрузитьНаСервере();

Процедура СоздатьДокументыНаСервере()

	ДокПоступлениеАвто.Дата = ТекущаяДата();

	ДокПоступлениеАвто.Записать();

	Сообщение.Текст = "Новый документ создан ! ";
	Сообщение.Сообщить();

//код для кнопки УдалитьДокумент
&НаСервере
Процедура УдалитьДокументНаСервере()
	НайденныйДокумент = Документы.ПоступлениеАвтомобилей.НайтиПоНомеру(НомерДокумента);
	Если НайденныйДокумент = Документы.ПоступлениеАвтомобилей.ПустаяСсылка () Тогда 
		Сообщение = Новый СообщениеПользователю;
		Сообщение.Текст = СтрШаблон ("Не найден документ под таким номером = %1", НомерДокумента);
		Сообщение.Сообщить();
	Иначе
		ОбъектДокумента = НайденныйДокумент.ПолучитьОбъект();
		Если УдалитьБезвозвратно Тогда
			ОбъектДокумента.Удалить();
			НайденныйДокумент = Неопределено;
			Сообщение = Новый СообщениеПользователю;
			Сообщение.Текст = " Документ " + НомерДокумента + " удален";
			Сообщение.Сообщить();
		Иначе
			ОбъектДокумента.ПометкаУдаления = Истина; 
			ОбъектДокумента.Записать (); 
			Сообщение = Новый СообщениеПользователю;
			Сообщение.Текст = " Документ " + НомерДокумента + " помечен на удаление";
			Сообщение.Сообщить();
		КонецЕсли;
	КонецЕсли;
