	Процедура ФайлExcelНачалоВыбора(Элемент, ДанныеВыбора, ВыборДобавлением, СтандартнаяОбработка)
		
		СтандартнаяОбработка = Ложь; 
		
		// Инициализация вызова диалога выбора файла с заданием параметров
		РежимВыбора = РежимДиалогаВыбораФайла.Открытие;
		ДиалогВыборФайлаExcel = Новый ДиалогВыбораФайла(РежимВыбора);
		ДиалогВыборФайлаExcel.МножественныйВыбор = Ложь;
		ДиалогВыборФайлаExcel.ПредварительныйПросмотр = Ложь;

		// Действия после выбора файла с помощью ДиалогаВыбораФайла
		ОписаниеОповещения = Новый ОписаниеОповещения("ПослеВыбораФайлаExcel", ЭтотОбъект);

	&НаКлиенте
	Асинх Процедура ЗагрузитьФайл(Команда)
		
		Сообщение = Новый СообщениеПользователю; 
		Если ЗначениеЗаполнено(ФайлExcel) Тогда 
			// получение параметров файла и перемещение на сервер для обработки
			ИнфоацияФайлExcel = Новый Файл(ФайлExcel);
			ОбещаниеПеремещенияФайла = Ждать ПоместитьФайлНаСерверАсинх( , , , ФайлExcel);
			Если ОбещаниеПеремещенияФайла <> Неопределено Тогда 
				// процедура обработки файла на сервере
				ЗагрузитьФайлНаСервере(ОбещаниеПеремещенияФайла.Адрес, ИнфоацияФайлExcel.Расширение);
				// вывод информации о загруженных элементах
				Сообщение.Текст = СтрШаблон("Загружено %1 элементов в таблицу", ДанныеExcel.Количество());
				Сообщение.Сообщить();
			КонецЕсли;
		Иначе
			Сообщение.Текст = "Вы не выбрали файл для загрузки";
			Сообщение.Поле = "ФайлExcel";
			Сообщение.Сообщить();
		КонецЕсли;
		
	КонецПроцедуры
	
	&НаКлиенте
	Процедура ЗаписатьВСправочник(Команда)
		
		Сообщение = Новый СообщениеПользователю;
		Если ДанныеExcel.Количество() > 0 Тогда 
			МассивНовыхАвтомобилей = Новый Массив;
			ЗаписатьВСправочникНаСервере(МассивНовыхАвтомобилей);
			ДобавленныеАвто = МассивНовыхАвтомобилей.Количество();  
			Сообщение.Текст = СтрШаблон (" Добавлено %1 элементов в Справочник", ДобавленныеАвто);
			Сообщение.Сообщить();
			
			// Вывод собщений о новых Автомобилей и обновление для форм
			ОтобразитьСозданныеЭлементы(МассивНовыхАвтомобилей);
			ОповеститьОбИзменении(Тип("СправочникСсылка.Автомобили"));
		Иначе 
			Сообщение.Текст = "Не заполнена таблица";
			Сообщение.Поле = "ДанныеExcel";
			Сообщение.Сообщить(); 
		КонецЕсли;
	КонецПроцедуры

	
#КонецОбласти

#Область СлужебныеПроцедурыИФункции

	&НаКлиенте
	Процедура ПослеВыбораФайлаExcel(ВыбранныеФайлы, ДополнительныеПараметры) Экспорт
		
		Если ВыбранныеФайлы = Неопределено Тогда
			Возврат;
		КонецЕсли;
		
		ФайлExcel = ВыбранныеФайлы[0];
		
	КонецПроцедуры

	&НаСервере
	Процедура ЗагрузитьФайлНаСервере(АдресХранилище, РасширениеФайла)
		
		// получение переданного файла на сервере и сохранения для обработки
		ИмяФайлаExcel = ПолучитьИмяВременногоФайла(РасширениеФайла); 
		ФайлExcelНаСервере = ПолучитьИзВременногоХранилища(АдресХранилище); 
	    ФайлExcelНаСервере.Записать(ИмяФайлаExcel) ; 
		
		// объявление Табличного документа и чтение в него данных из файла
		ТабДокExcel = Новый ТабличныйДокумент;
		ТабДокExcel.Прочитать(ИмяФайлаExcel, СпособЧтенияЗначенийТабличногоДокумента.Текст);
		
		// получение только первого листа из всех областей
		ИмяПервогоЛиста = ТабДокExcel.Области[0].Имя;
		ОбластьЧтения = ТабДокExcel.Область(ИмяПервогоЛиста);
		
		// Получение данных с помощью построителя запроса
		ПостроительЗапроса = Новый ПостроительЗапроса;
		ПостроительЗапроса.ИсточникДанных = Новый ОписаниеИсточникаДанных(ОбластьЧтения);
		ПостроительЗапроса.Выполнить();
		ТаблицаExcel = ПостроительЗапроса.Результат.Выгрузить();
		
		// Функция соответствия типов кузова, для логики обработки данных
		ТипыКузова = СоответствиеТипаКузова();
		ДанныеExcel.Очистить();
		
		// обработка полученных строк
		Для Каждого Эл Из ТаблицаExcel Цикл 
			НоваяСтрокаТЗ = ДанныеExcel.Добавить();
			ЗаполнитьЗначенияСвойств(НоваяСтрокаТЗ, Эл);
			НоваяСтрокаТЗ.Кузов = ТипыКузова[ВРег(Эл.Кузов)]; 
		КонецЦикла;
		
		// принудительное удаление созданного файла на сервере
		УдалитьФайлы(ИмяФайлаExcel);
		
	КонецПроцедуры

    &НаСервереБезКонтекста
	Функция СоответствиеТипаКузова()
		
		// с помощью запроса получем значения типов кузова и задаем соответствие
		РезультатФункции = Новый Соответствие;
		Запрос = Новый Запрос;
		Запрос.Текст = 
			"ВЫБРАТЬ
			|	ТипКузова.Наименование КАК Наименование,
			|	ТипКузова.Ссылка КАК Ссылка
			|ИЗ
			|	Справочник.ТипКузова КАК ТипКузова";
		РезультатЗапроса = Запрос.Выполнить();
		Выборка = РезультатЗапроса.Выбрать();
		Пока Выборка.Следующий() Цикл	
			РезультатФункции.Вставить(ВРег(Выборка.Наименование), Выборка.Ссылка);	
		КонецЦикла;
		Возврат РезультатФункции;
		
	КонецФункции 
	
	&НаСервере
	Процедура ЗаписатьВСправочникНаСервере(МассивНовыхАвтомобилей)
		
		Для Каждого Эл Из ДанныеExcel Цикл 
			
			ПроверкаАвто = Справочники.Автомобили.НайтиПоНаименованию(Эл.Наименование);
			Если ПроверкаАвто = Справочники.Автомобили.ПустаяСсылка() Тогда
				НовыйЭлементСправочника = Справочники.Автомобили.СоздатьЭлемент();
				ЗаполнитьЗначенияСвойств(НовыйЭлементСправочника, Эл);
				НовыйЭлементСправочника.Записать();
				МассивНовыхАвтомобилей.Добавить(НовыйЭлементСправочника.Ссылка); 
			Иначе 
				Сообщение = Новый СообщениеПользователю;
				Сообщение.Текст = " Такой элемент " + Эл.Наименование + " уже существует в справочнике, он не был добавлен повторно";
				Сообщение.Сообщить();
		КонецЕсли; 
		КонецЦикла;
		
	КонецПроцедуры

	&НаСервере
	Процедура УсловноеОформлениеФормы()
		
		// программная установка условного оформления для элементов формы
		СоответствиеЦветов = СоответствиеЦветов();
		ЭлементыУсловногоОформления = УсловноеОформление.Элементы;
		Для Каждого Эл Из СоответствиеЦветов Цикл
			ЭлементОформления = ЭлементыУсловногоОформления.Добавить();
			ЭлементОтбора = ЭлементОформления.Отбор.Элементы.Добавить(Тип("ЭлементОтбораКомпоновкиДанных"));
			ЭлементОтбора.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("ДанныеExcel.Цвет");
			ЭлементОтбора.ВидСравнения = ВидСравненияКомпоновкиДанных.Равно;
			ЭлементОтбора.ПравоеЗначение = Эл.Ключ;
			ЭлементОтбора.Использование = Истина;
			ЭлементОформления.Оформление.УстановитьЗначениеПараметра("ЦветФона", Эл.Значение); 
			ПолеОформления = ЭлементОформления.Поля.Элементы.Добавить();
			ПолеОформления.Поле = Новый ПолеКомпоновкиДанных("ДанныеExcelЦвет");
			ПолеОформления.Использование = Истина;
			ЭлементОформления.Использование = Истина; 	
		КонецЦикла;
