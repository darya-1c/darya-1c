		|СГРУППИРОВАТЬ ПО

		|ИНДЕКСИРОВАТЬ ПО

	Движение.Период = Дата;
	
	Движение.Склад = Склад;
	
	Движение.Автомобиль = ВыборкаАвтомобили.Автомобиль;
	
	Движение.Количество = ВыборкаАвтомобили.Количество;
	
	КонецЦикла;

	Движения.КоличествоАвтомобилей.Записывать = Истина;

     Запрос.Текст =

		|	ОстаткиАвтомобилей.Автомобиль КАК Автомобиль,
